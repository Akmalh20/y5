Google Sheets Integration & Student Records

Google Apps Script Setup Instructions:
  1. Buka Google Sheet baharu & pergi ke Extensions > Apps Script.
  2. Tampal kod Apps Script di bawah dan tekan Deploy > New deployment.
  3. Pilih Web app, tetapkan Access kepada "Anyone", kemudian Deploy.
  4. Salin URL Web App dan tampal ke dalam ruang di bawah.

Google Apps Script Code Snippet: 
function doPost(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var data = JSON.parse(e.postData.contents);
  
  if (sheet.getLastRow() === 0) {
    sheet.appendRow(["Timestamp", "Player Name", "Team", "Total Score", "Goals Scored", "Sentences Correct", "Accuracy %"]);
  }
  
  sheet.appendRow([
    new Date(),
    data.playerName,
    data.team,
    data.totalScore,
    data.goalsScored,
    data.sentencesCorrect,
    data.accuracy + "%"
  ]);
  
  return ContentService.createTextOutput(JSON.stringify({"result": "success"})).setMimeType(ContentService.MimeType.JSON);
}

function doGet() {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var rows = sheet.getDataRange().getValues();
  var result = [];
  var headers = rows[0];
  
  for (var i = 1; i < rows.length; i++) {
    var obj = {};
    for (var j = 0; j < headers.length; j++) {
      obj[headers[j]] = rows[i][j];
    }
    result.push(obj);
  }
  return ContentService.createTextOutput(JSON.stringify(result)).setMimeType(ContentService.MimeType.JSON);
}
